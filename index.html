async function sendCommand() { const createdAt = Date.now(); const durationMinutes = Number(document.getElementById("duration").value); const command = { targetType: document.getElementById("targetType").value, target: document.getElementById("target").value, action: document.getElementById("action").value, // Stored as minutes duration: durationMinutes, // UTC timestamp in milliseconds createdAt: createdAt, // Exact time the action expires expiresAt: createdAt + (durationMinutes * 60 * 1000), reason: document.getElementById("reason").value }; await fetch(FIREBASE_DATABASE + COMMAND_LOCATION, { method:"POST", headers:{ "Content-Type":"application/json" }, body: JSON.stringify(command) }); await fetch(FIREBASE_DATABASE + LOG_LOCATION, { method:"POST", headers:{ "Content-Type":"application/json" }, body: JSON.stringify({ action: command.action, target: command.target, reason: command.reason, createdAt: createdAt }) }); if(command.action === "ban") { await fetch(FIREBASE_DATABASE + HISTORY_LOCATION, { method:"POST", headers:{ "Content-Type":"application/json" }, body: JSON.stringify(command) }); } document.getElementById("status").innerText = "Command sent to Roblox system."; }
